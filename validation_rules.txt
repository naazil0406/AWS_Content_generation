[EMPTY_INPUT]
warning = Please enter a prompt before generating content.

[SPECIAL_CHARACTERS]
# Applies only when the ENTIRE prompt has no letters or digits at all
# (e.g. "@", "!!!", "@#$%"). Prompts that mix punctuation with real
# words (e.g. "AI & Cybersecurity", "COVID-19 safety") are never
# affected by this rule.
warning = ⚠️ Invalid prompt: Special characters alone are not allowed. Please enter meaningful words.

[NUMBERS_ONLY]
# Applies only when the ENTIRE prompt is digits (e.g. "123456", "2026",
# "5"). Numbers inside a meaningful prompt ("Create 5 safety tips") are
# never affected by this rule.
warning = ⚠️ Invalid prompt: Numbers alone are not a meaningful content request. Please describe what you'd like created.

[MEANINGFUL_WORDS]
# Covers keyboard-mash (asdfgh, qwerty), single-repeated-character runs
# (aaaaaaa, zzzzzzz), immediately-repeated-substring input
# (testtesttest), generic greetings/small talk ("hello", "thanks", "ok")
# that name no topic, and anything else the LLM-based semantic check
# judges to specify no identifiable subject/topic — see
# app/services/validation_service.py's _ANALYSIS_SYSTEM_PROMPT for the
# full, generalized (never hardcoded to one example) definition.
warning = Please provide a specific topic or content-generation request.
fallback_examples = For example: "Create an infographic about [topic]", "Create a poster about [topic]", "Create [number] tips about [topic]", or "Create a safety awareness campaign about [topic]".

[SPELLING]

[CONTENT_MODERATION]
# Shown for ANY blocked request — harmful, dangerous, sexual/explicit,

# model; this file only controls the user-facing MESSAGE, not the
# detection logic itself, which must stay in code/the model to remain

[KNOWLEDGE_BASE_RELEVANCE]
